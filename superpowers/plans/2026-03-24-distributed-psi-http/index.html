    Distributed PSI over HTTP Implementation Plan¶

    For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

    Goal: Split compute_cn across two real network services so Party 1 and Party 2 can run on separate machines, with each party holding only their own graph.

    Architecture: Party 2 runs a FastAPI server that exposes endpoints for the PSI protocol handshake — it holds Graph 2 in memory, computes its neighbor sets, and responds to PSI setup/respond round-trips. Party 1 holds Graph 1, orchestrates the 3 PSI calls over HTTP via compute_cn_remote, and is the only party that learns the final CN score. The existing local compute_cn(graph1, graph2, x, y) is kept unchanged for tests and demos. Graph and session store are held on app.state (not module globals) so multiple app instances can coexist in the same process during tests.

    Tech Stack: FastAPI, Uvicorn, HTTPX (sync HTTP client), openmined-psi (protobuf messages serialized as base64 over JSON), private_set_intersection.proto.psi_pb2 for deserialization.

    Serialization Contract¶

    Each PSI call requires two HTTP round-trips. Protobuf objects are serialized as base64-encoded strings in JSON bodies:

    setup_b64   = base64.b64encode(setup.SerializeToString()).decode()
setup_proto = ServerSetup.FromString(base64.b64decode(setup_b64))

    Same pattern for Request and Response.

    API: Party 2 Endpoints¶

    POST /prepare
  Body:     { "x": str, "y": str }
  Response: { "direct_link": bool, "local2": int, "session_id": str }

POST /psi/{session_id}/setup
  Body:     { "call": "crossover1"|"crossover2"|"overlap", "client_set_size": int }
  Response: { "setup": str }   # base64 ServerSetup protobuf

POST /psi/{session_id}/respond
  Body:     { "call": "crossover1"|"crossover2"|"overlap", "request": str }  # base64 Request
  Response: { "response": str }  # base64 Response protobuf

    For each call: - crossover1: Party 2 uses g2_y - local2_set - crossover2: Party 2 uses g2_x - local2_set - overlap: Party 2 uses local2_set

    Known limitation: Sessions are held in memory and never garbage-collected for the lifetime of the server process. For a class demo with low query volume this is fine.

    from __future__ import annotations
import uuid
from dataclasses import dataclass, field
from typing import Any


@dataclass
class PsiSession:
    sets: dict[str, set[str]]     # call -> Party 2's neighbor set for that call
    local2: int                    # |Γ₂(x) ∩ Γ₂(y)| — revealed to Party 1 per protocol
    servers: dict[str, Any] = field(default_factory=dict)  # call -> psi.server object


class SessionStore:
    def __init__(self):
        self._store: dict[str, PsiSession] = {}

    def put(self, session: PsiSession) -> str:
        sid = str(uuid.uuid4())
        self._store[sid] = session
        return sid

    def get(self, session_id: str) -> PsiSession | None:
        return self._store.get(session_id)

    def delete(self, session_id: str) -> None:
        self._store.pop(session_id, None)

    from __future__ import annotations
import base64
from typing import Literal

import private_set_intersection.python as psi
from fastapi import APIRouter, HTTPException, Request
from pydantic import BaseModel

from pplp.server.session import PsiSession

_FPR = 0.001
_DATA_STRUCTURE = psi.DataStructure.GCS
CallType = Literal["crossover1", "crossover2", "overlap"]

router = APIRouter()


# --- /prepare ---

class PrepareRequest(BaseModel):
    x: str
    y: str


class PrepareResponse(BaseModel):
    direct_link: bool
    local2: int
    session_id: str


@router.post("/prepare", response_model=PrepareResponse)
def prepare(req: PrepareRequest, request: Request):
    g2 = request.app.state.graph
    store = request.app.state.store

    x, y = req.x, req.y
    if g2.has_edge(x, y):
        return PrepareResponse(direct_link=True, local2=0, session_id="")

    local2_set = g2.local_intersection(x, y)
    g2_x = g2.neighbors(x) - local2_set
    g2_y = g2.neighbors(y) - local2_set

    session = PsiSession(
        sets={
            "crossover1": g2_y,       # Party 2's set for crossover1 = g2_y (excl. local2)
            "crossover2": g2_x,       # Party 2's set for crossover2 = g2_x (excl. local2)
            "overlap":    local2_set, # Party 2's set for overlap
        },
        local2=len(local2_set),
    )
    sid = store.put(session)
    return PrepareResponse(direct_link=False, local2=session.local2, session_id=sid)

    from private_set_intersection.proto.psi_pb2 import Request as PsiRequest


class PsiSetupRequest(BaseModel):
    call: CallType
    client_set_size: int


class PsiSetupResponse(BaseModel):
    setup: str      # base64 ServerSetup


class PsiRespondRequest(BaseModel):
    call: CallType
    request: str    # base64 Request


class PsiRespondResponse(BaseModel):
    response: str   # base64 Response


@router.post("/psi/{session_id}/setup", response_model=PsiSetupResponse)
def psi_setup(session_id: str, req: PsiSetupRequest, request: Request):
    store = request.app.state.store
    session = store.get(session_id)
    if session is None:
        raise HTTPException(status_code=404, detail="Session not found")

    party2_set = list(session.sets[req.call])
    server = psi.server.CreateWithNewKey(reveal_intersection=False)
    setup = server.CreateSetupMessage(_FPR, req.client_set_size, party2_set, _DATA_STRUCTURE)
    session.servers[req.call] = server

    return PsiSetupResponse(setup=base64.b64encode(setup.SerializeToString()).decode())


@router.post("/psi/{session_id}/respond", response_model=PsiRespondResponse)
def psi_respond(session_id: str, req: PsiRespondRequest, request: Request):
    store = request.app.state.store
    session = store.get(session_id)
    if session is None:
        raise HTTPException(status_code=404, detail="Session not found")

    server = session.servers.get(req.call)
    if server is None:
        raise HTTPException(status_code=400, detail=f"Setup not called for call '{req.call}'")

    request_proto = PsiRequest.FromString(base64.b64decode(req.request))
    response = server.ProcessRequest(request_proto)
    return PsiRespondResponse(response=base64.b64encode(response.SerializeToString()).decode())

    from __future__ import annotations
import base64

import private_set_intersection.python as psi
from private_set_intersection.proto.psi_pb2 import ServerSetup, Response as PsiResponse

_FPR = 0.001


def remote_psi_cardinality(
    http_client,
    client_set: set[str],
    session_id: str,
    call: str,
) -> int:
    """Perform one PSI-cardinality call against Party 2's server.

    Args:
        http_client: An httpx.Client pointed at Party 2's base URL,
                     or a FastAPI TestClient (same interface).
        client_set:  Party 1's set for this PSI call.
        session_id:  Session ID returned by POST /prepare.
        call:        One of "crossover1", "crossover2", "overlap".
    """
    if not client_set:
        return 0

    psi_client = psi.client.CreateWithNewKey(reveal_intersection=False)
    client_list = list(client_set)

    setup_resp = http_client.post(
        f"/psi/{session_id}/setup",
        json={"call": call, "client_set_size": len(client_list)},
    )
    setup_resp.raise_for_status()
    setup = ServerSetup.FromString(base64.b64decode(setup_resp.json()["setup"]))

    request = psi_client.CreateRequest(client_list)
    request_b64 = base64.b64encode(request.SerializeToString()).decode()
    respond_resp = http_client.post(
        f"/psi/{session_id}/respond",
        json={"call": call, "request": request_b64},
    )
    respond_resp.raise_for_status()
    response = PsiResponse.FromString(base64.b64decode(respond_resp.json()["response"]))

    return psi_client.GetIntersectionSize(setup, response)

    def compute_cn_remote(graph1: Graph, party2_client, x: str, y: str) -> int:
    """Distributed compute_cn — Party 1 holds graph1, Party 2 is a remote HTTP service.

    Args:
        graph1:        Party 1's graph.
        party2_client: An httpx.Client or FastAPI TestClient pointed at Party 2's server.
        x, y:          The candidate node pair (must match node IDs used in both graphs).
    """
    from pplp.psi_client import remote_psi_cardinality

    if graph1.has_edge(x, y):
        raise ValueError(
            f"{x} and {y} are direct neighbors in graph1; "
            "no need for the computation"
        )

    prep_resp = party2_client.post("/prepare", json={"x": x, "y": y})
    prep_resp.raise_for_status()
    prep = prep_resp.json()

    if prep["direct_link"]:
        raise DirectLinkFound(f"{x} and {y} are direct neighbors in graph2")

    local2 = prep["local2"]
    session_id = prep["session_id"]

    local1_set = graph1.local_intersection(x, y)
    local1 = len(local1_set)

    g1_x = graph1.neighbors(x) - local1_set
    g1_y = graph1.neighbors(y) - local1_set

    crossover1 = remote_psi_cardinality(party2_client, g1_x, session_id, "crossover1")
    crossover2 = remote_psi_cardinality(party2_client, g1_y, session_id, "crossover2")
    overlap = remote_psi_cardinality(party2_client, local1_set, session_id, "overlap")

    return local1 + local2 + crossover1 + crossover2 - overlap

import csv

from fastapi import FastAPI

def create_app(graph: Graph) -> FastAPI:
    app = FastAPI(title="PPLP Party 2 Server")
    app.state.graph = graph
    app.state.store = SessionStore()

    from pplp.server.routes import router
    app.include_router(router)

    return app


def load_graph_from_csv(path: str) -> Graph:
    """Load a graph from a two-column CSV (no header): one edge per row as 'u,v'."""
    edges = []
    with open(path, newline="") as f:
        for row in csv.reader(f):
            if len(row) >= 2:
                edges.append((row[0].strip(), row[1].strip()))
    return Graph.from_edge_list(edges)

    args = parser.parse_args()

    graph = load_graph_from_csv(args.graph_file)
    app = create_app(graph)
    uvicorn.run(app, host=args.host, port=args.port)

    Privacy Notes (for writeup)¶

    • Party 2 learns: x, y, |client_set| for each PSI call (via client_set_size), and the size of its own local intersection. This is the intentional semi-honest leakage described in Ayday et al.
    • Party 1 learns: local2 (explicit from /prepare), and the three PSI intersection sizes. Party 1 never sees Party 2's raw adjacency list.
    • client_set_size trust: Under the semi-honest model, Party 1 is trusted to send the correct set size. A dishonest Party 1 could skew the Bloom filter sizing, but this is out-of-scope per the paper's threat model.
    • Session cleanup: Sessions are never deleted from memory. Not a concern for a demo; a production system would add a TTL.
